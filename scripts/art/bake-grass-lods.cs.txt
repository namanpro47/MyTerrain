var results=new System.Collections.Generic.List<object>();
foreach(var name in new[]{"Grass_Common_Short","Grass_Wispy_Short","Grass_Common_Tall"}){
var go=UnityEngine.Resources.Load<UnityEngine.GameObject>("Nature/"+name);var filters=go.GetComponentsInChildren<UnityEngine.MeshFilter>(true);

}}}
UnityEditor.AssetDatabase.SaveAssets();return results;
