Shader "Terra/Studies/ContinuousGrain52"
{
 Properties {
  _GrainColor("Artist grain albedo",Color)=(.72,.62,.34,1)
  _MeadowPaint("Artist brush texture",2D)="white"{}
  _LandCoverMap("Cover",2D)="black"{}
  _TreatmentMap("Treatment",2D)="black"{}
  _RiverMap("River",2D)="white"{}
  _PathMap("Path",2D)="white"{}
  _LandBounds("Bounds",Vector)=(-590,-630,1180,1260)
  _CanopyHeight("Canopy height",Float)=.95
 }
 SubShader {
  Tags {"RenderType"="TransparentCutout" "Queue"="AlphaTest" "RenderPipeline"="UniversalPipeline"}
  Pass {
   Tags {"LightMode"="UniversalForward"}
   Cull Off
   HLSLPROGRAM
   #pragma vertex Vert
   #pragma fragment Frag
   #pragma multi_compile _ _MAIN_LIGHT_SHADOWS _MAIN_LIGHT_SHADOWS_CASCADE
   #pragma multi_compile_fragment _ _SHADOWS_SOFT _SHADOWS_SOFT_LOW _SHADOWS_SOFT_MEDIUM _SHADOWS_SOFT_HIGH
   #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
   #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Lighting.hlsl"
   CBUFFER_START(UnityPerMaterial)
    half4 _GrainColor;float4 _LandBounds;float _CanopyHeight;
   CBUFFER_END
   TEXTURE2D(_MeadowPaint);SAMPLER(sampler_MeadowPaint);
   TEXTURE2D(_LandCoverMap);SAMPLER(sampler_LandCoverMap);
   TEXTURE2D(_TreatmentMap);SAMPLER(sampler_TreatmentMap);
   TEXTURE2D(_RiverMap);SAMPLER(sampler_RiverMap);
   TEXTURE2D(_PathMap);SAMPLER(sampler_PathMap);
   struct A {float4 positionOS:POSITION;};
   struct V {float4 positionCS:SV_POSITION;float3 positionWS:TEXCOORD0;};
   V Vert(A i){V o;o.positionWS=TransformObjectToWorld(i.positionOS.xyz);o.positionWS.y+=_CanopyHeight+sin(o.positionWS.x*.3+o.positionWS.z*.21)*.08;o.positionCS=TransformWorldToHClip(o.positionWS);return o;}
   half4 Frag(V i):SV_Target {
    float2 p=i.positionWS.xz;float2 uv=(p-_LandBounds.xy)/_LandBounds.zw;
    half4 cover=SAMPLE_TEXTURE2D(_LandCoverMap,sampler_LandCoverMap,uv);
    half4 treat=SAMPLE_TEXTURE2D(_TreatmentMap,sampler_TreatmentMap,uv);
    half field=lerp(cover.g,treat.r/max(treat.r+treat.g,.0001),treat.a);
    clip(field-.54);
    clip(SAMPLE_TEXTURE2D(_RiverMap,sampler_RiverMap,uv).r-.43);
    clip(SAMPLE_TEXTURE2D(_PathMap,sampler_PathMap,uv).r-.75);
    float fieldU=p.x-12*sin(p.y*.012)-5*sin(p.y*.029);
    float track=abs(frac((fieldU+19.2)/38.4)*38.4-19.2);
    clip(track-.70);
    float patch=.5+.5*sin(p.x*.075+sin(p.y*.043)*2.5);
    half3 grain=lerp(_GrainColor.rgb*half3(.86,.85,.7),_GrainColor.rgb*half3(1.08,1.06,1.04),patch);
    half3 paint=SAMPLE_TEXTURE2D(_MeadowPaint,sampler_MeadowPaint,p/8).rgb;
    float brush=clamp(dot(paint,half3(.2126,.7152,.0722))/.18,.8,1.2);
    float row=.5+.5*sin(fieldU*4.654211);
    float detail=1-smoothstep(.12,.4,max(length(ddx(p)),length(ddy(p))));
    Light sun=GetMainLight(TransformWorldToShadowCoord(i.positionWS));
    half3 light=SampleSH(half3(0,1,0))+sun.color*.85*lerp(.5,1,sun.shadowAttenuation);
    return half4(grain*light*.90*lerp(1,brush,.45)*(1-row*.06*detail),1);
   }
   ENDHLSL
  }
 }
}
