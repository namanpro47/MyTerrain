 foreach(float side in new[]{-1f,1f})
 {
  var v=new System.Collections.Generic.List<UnityEngine.Vector3>();var uv=new System.Collections.Generic.List<UnityEngine.Vector2>();var wall=new System.Collections.Generic.List<int>();var cap=new System.Collections.Generic.List<int>();float along=0;
  for(int i=0;i<samples.Count;i++)
  {

   if(i==0)continue;

  }

  var go=new UnityEngine.GameObject("Bank "+side);go.transform.SetParent(holder.transform,false);go.AddComponent<UnityEngine.MeshFilter>().sharedMesh=mesh;go.AddComponent<UnityEngine.MeshRenderer>().sharedMaterials=new[]{material,ground};vertices+=v.Count;triangles+=(wall.Count+cap.Count)/3;
 }
