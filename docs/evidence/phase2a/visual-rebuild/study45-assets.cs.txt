var rows=new System.Collections.Generic.List<object>();int checks=0;void Check(bool value,string reason){checks++;if(!value)throw new System.Exception(reason);}
foreach(var id in new[]{1,3,5}){var p=UnityEngine.Resources.Load<UnityEngine.GameObject>("ClusterForest/ClusterTree_"+id);Check(p!=null,"Missing prefab");Check(p.transform.localScale==UnityEngine.Vector3.one&&p.transform.localRotation==UnityEngine.Quaternion.identity,"Nonmetric root");var lods=p.GetComponent<UnityEngine.LODGroup>().GetLODs();Check(lods.Length==3,"Missing distance levels");var source=lods[0].renderers[0].GetComponent<UnityEngine.MeshFilter>().sharedMesh;var meshes=new System.Collections.Generic.List<object>();for(int level=0;level<lods.Length;level++){Check(lods[level].renderers.Length==1,"Mismatched renderer correspondence");var r=lods[level].renderers[0];var m=r.GetComponent<UnityEngine.MeshFilter>().sharedMesh;Check(m.subMeshCount==r.sharedMaterials.Length&&m.subMeshCount==2,"Missing submesh/material");Check(r.sharedMaterials.All(x=>x!=null&&x.shader.isSupported&&x.enableInstancing),"Unsupported/noninstanced material");Check(m.normals.All(n=>!float.IsNaN(n.x)&&!float.IsNaN(n.y)&&!float.IsNaN(n.z)&&n.sqrMagnitude>.98f&&n.sqrMagnitude<1.02f),"Invalid normals");Check(m.triangles.All(i=>i>=0&&i<m.vertexCount),"Invalid geometry index");Check(m.uv.Length==m.vertexCount,"Lost source UVs");Check(m.GetIndexCount(0)>0&&m.GetIndexCount(1)>0,"Lost wood or foliage");Check(UnityEngine.Mathf.Abs(m.bounds.min.y)<.02f,"Root moved above ground");Check(m.bounds.size.y/source.bounds.size.y>.95f&&m.bounds.size.y/source.bounds.size.y<1.12f,"Crown height changed materially");if(level>0)Check(m.triangles.Length<source.triangles.Length*.6f,"Insufficient geometry reduction");meshes.Add(new{level,triangles=m.triangles.Length/3,vertices=m.vertexCount,height=m.bounds.size.y});}rows.Add(new{name=p.name,meshes});}
return new{checks,rows};
